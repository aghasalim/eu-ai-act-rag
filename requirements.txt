# Core pipeline. No LLM vendor SDKs: every supported provider (Groq, OpenAI,
# Gemini, Ollama) speaks the OpenAI-compatible chat API over plain HTTP.

# Prefer CPU-only torch on Linux. PEP 440 sorts the local version "2.x+cpu"
# above plain "2.x", so on Linux pip picks the CPU wheel from this index instead
# of dragging ~2.5 GB of CUDA libraries down from PyPI: which is what blows the
# disk budget on free hosts. macOS/Windows have no "+cpu" wheels here, so pip
# falls through to PyPI and nothing changes locally.
--extra-index-url https://download.pytorch.org/whl/cpu

# chromadb needs sqlite3 >= 3.35; the Debian image on Streamlit Community Cloud
# ships an older one. app/streamlit_app.py swaps this in under the stdlib name.
pysqlite3-binary; sys_platform == "linux"
